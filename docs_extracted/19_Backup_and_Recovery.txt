#!/usr/bin/env bash # /usr/local/bin/backup_audio.sh set -euo pipefail  BACKUP_DIR="/var/backups/audio" TIMESTAMP=$(date +"%Y%m%d_%H%M%S") mkdir -p "$BACKUP_DIR"  # 1. Dump MySQL Database và nén gzip tức thì mysqldump --single-transaction --quick -u root audio_db | gzip -9 > "$BACKUP_DIR/db_$TIMESTAMP.sql.gz"  # 2. Đồng bộ tệp MP3 và cover sang Cloud Storage rẻ (R2 / S3) qua rclone rclone sync /data/audio/ remote_r2:audio-backup-bucket/data/ --fast-list  # 3. Đẩy file dump DB lên cloud rclone copy "$BACKUP_DIR/db_$TIMESTAMP.sql.gz" remote_r2:audio-backup-bucket/db/  # 4. Giữ lại bản sao lưu cục bộ tối đa 7 ngày find "$BACKUP_DIR" -type f -name "*.sql.gz" -mtime +7 -delete
