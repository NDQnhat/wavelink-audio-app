// Regex nhận dạng chính xác Video ID 11 ký tự từ mọi dạng link YouTube phổ biến var ytRegex = regexp.MustCompile(`^(?:https?:\/\/)?(?:www\.|m\.)?(?:youtube\.com\/(?:watch\?v=|embed\/|v\/|shorts\/)|youtu\.be\/)([a-zA-Z0-9_-]{11})(?:[?&].*)?$`)  func ExtractVideoID(rawURL string) (string, error) {     if len(rawURL) > 500 {         return "", errors.New("URL quá dài")     }     matches := ytRegex.FindStringSubmatch(strings.TrimSpace(rawURL))     if len(matches) < 2 {         return "", errors.New("URL không hợp lệ hoặc không phải YouTube Video")     }     videoID := matches[1]     return videoID, nil }
